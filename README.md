```python
import sys
import time

def coding_animation():
    frames = [
        "🚀 Writing Code...   ",
        "🔍 Debugging...      ",
        "🔥 Optimizing...     ",
        "✅ Deployed!         "
    ]
    for _ in range(3):  # Loop animation for effect
        for frame in frames:
            sys.stdout.write("\r" + frame)
            sys.stdout.flush()
            time.sleep(0.5)
    print("\nDone!")

coding_animation()
```
